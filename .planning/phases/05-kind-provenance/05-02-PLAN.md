      it('16 real raws exceed MAX_TOTAL_CHARS', () => {
        const RAW_BYTES = 17_055_535  // one 8-direction raw strip
        const RAW_COUNT = 16

        const totalBase64Chars = RAW_BYTES * RAW_COUNT * BASE64_EXPANSION
        
        // MAX_TOTAL_CHARS from route.ts:15
        const MAX_TOTAL_CHARS = Math.ceil((200 * 1024 * 1024 * 4) / 3) + 1024
        
        expect(totalBase64Chars).toBeGreaterThan(MAX_TOTAL_CHARS)
        // 363.8M > 279.6M → raw in structural impossibility

        const MAX_TOTAL_CHARS = Math.ceil((200 * 1024 * 1024 * 4) / 3) + 1024

        // 72M < 279.6M → derived-only is ~26% of cap
      })
